! =========================================================================
! SCRIPT CẤU HÌNH SWITCH MỞ RỘNG KHU B: SW-LAB-02 (Cisco Catalyst 2960-24TT)
! ĐỀ TÀI 1: MẠNG LAN CHO PHÒNG THỰC HÀNH 30 MÁY (SUBNET 192.168.10.0/26)
! Vai trò: Kết nối 15 PC Khu B (PC16 -> PC30 trên Fa0/1 -> Fa0/15),
!          Liên kết Trunk tốc độ cao về SW-LAB-01 qua cổng Gi0/1.
! Hướng dẫn: Mở SW-LAB-02 -> Tab CLI -> Nhấn Enter -> Copy toàn bộ & Paste!
! =========================================================================

enable
configure terminal

! 1. Đặt tên thiết bị và tắt tra cứu DNS khi gõ nhầm lệnh
hostname SW-LAB-02
no ip domain-lookup

! 2. Cấu hình mật khẩu quản trị và mã hóa mật khẩu
enable secret DaiNam@2026
service password-encryption

! 3. Cấu hình thông điệp cảnh báo đăng nhập (Banner MOTD)
banner motd #
***********************************************************************
* CANH BAO: THIET BI CHUYEN MACH MO RONG SW-LAB-02 (KHU VUC B)        *
* PHONG THUC HANH MAY TINH 30 MAY - KHOA CNTT - DAI HOC DAI NAM       *
* MOI TRUY CAP TRAI PHEP SE BI GHI LOG VA XU LY THEO QUY DINH!        *
***********************************************************************
#

! 4. Bảo mật cổng Console và cổng VTY (Telnet/SSH)
line console 0
 password Console@123
 login
 logging synchronous
 exec-timeout 15 0
 exit

line vty 0 15
 password Telnet@123
 login
 logging synchronous
 exec-timeout 15 0
 transport input telnet ssh
 exit

! 5. Cấu hình IP quản trị (Management IP) trên SVI VLAN 1 và Default Gateway
interface Vlan1
 description MANAGEMENT_SVI_SW-LAB-02
 ip address 192.168.10.3 255.255.255.192
 no shutdown
 exit

ip default-gateway 192.168.10.1

! 6. Cấu hình dải cổng kết nối 15 Máy trạm Khu B (PC16 -> PC30: Fa0/1 -> Fa0/15)
interface range FastEthernet0/1 - 15
 description ACCESS_PORTS_KHU_B_PC16_TO_PC30
 switchport mode access
 switchport access vlan 1
 spanning-tree portfast
 no shutdown
 exit

! 7. Tắt (Shutdown) các cổng dự phòng không sử dụng (Fa0/16 -> Fa0/24 và Gi0/2)
interface range FastEthernet0/16 - 24
 description UNUSED_PORTS_SHUTDOWN_FOR_SECURITY
 switchport mode access
 shutdown
 exit

interface GigabitEthernet0/2
 description UNUSED_GIGABIT_PORT_SHUTDOWN
 switchport mode access
 shutdown
 exit

! 8. Cấu hình cổng Trunk nối về SW-LAB-01 (Gi0/1)
interface GigabitEthernet0/1
 description TRUNK_UPLINK_TO_SW-LAB-01_Gi0/2
 switchport mode trunk
 no shutdown
 exit

! 9. Lưu cấu hình vào NVRAM
end
copy running-config startup-config

! Kiểm tra nhanh trạng thái sau khi cấu hình
show ip interface brief
show interfaces trunk
